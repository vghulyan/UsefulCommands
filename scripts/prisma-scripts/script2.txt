#!/bin/bash

# Recursively scan for .crt or .pem files
find . -type f \( -iname "*.crt" -o -iname "*.pem" \) | while read -r cert; do
  # Extract subject and issuer, remove leading labels
  subj=$(openssl x509 -in "$cert" -noout -subject 2>/dev/null | sed 's/^subject= //')
  issu=$(openssl x509 -in "$cert" -noout -issuer 2>/dev/null | sed 's/^issuer= //')

  # Check if cert is self-signed (issuer == subject)
  if [ "$subj" = "$issu" ]; then
    echo "🔐 Root CA found: $cert"
    echo "  Subject: $subj"
    echo
  fi
done
